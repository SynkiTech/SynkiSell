    32→    32→export function PosView({ businessId, onShiftChanged }: { businessId: string; onShiftChanged?: () => void }) {
    33→    33→  const { data: session } = useSession();
    34→    34→  const canDiscountLine = session?.user?.role === "OWNER" || session?.user?.role === "MANAGER" || session?.user?.role === "PHARMACIST";
    35→    35→
    36→    36→  const { cart, addToCart, updateCartQty, updateCartDiscount, removeFromCart, clearCart, offline, setOffline, bumpRefresh } = useAppStore();
    37→    37→  const [catalog, setCatalog] = useState<CatalogCache | null>(null);
    38→    38→  const [search, setSearch] = useState("");
    39→    39→  const [deptFilter, setDeptFilter] = useState<"ALL" | "PHARMACY" | "SUPERMARKET">("ALL");
    40→    40→  const [shift, setShift] = useState<{ id: string; openingCash: number } | null>(null);
    41→    41→  const [openShiftDialog, setOpenShiftDialog] = useState(false);
    42→    42→  const [openingFloat, setOpeningFloat] = useState("5000");
    43→    43→  const [payDialog, setPayDialog] = useState(false);
    44→    44→  const [paymentMethod, setPaymentMethod] = useState<"CASH" | "TRANSFER" | "CARD">("CASH");
    45→    45→  const [cashGiven, setCashGiven] = useState("");
    46→    46→  const [transferRef, setTransferRef] = useState("");
    47→    47→  const [splitCash, setSplitCash] = useState("");
    48→    48→  const [splitTransfer, setSplitTransfer] = useState("");
    49→    49→  const [useSplit, setUseSplit] = useState(false);
    50→    50→  const [completing, setCompleting] = useState(false);
    51→    51→  const [receipt, setReceipt] = useState<Receipt | null>(null);
    52→    52→  const [offlineWarnings, setOfflineWarnings] = useState<string[]>([]);
    53→    53→  const barcodeRef = useRef<HTMLInputElement>(null);
    54→    54→
    55→    55→  const total = cartTotal(cart);
    56→    56→  const { online } = offline;
    57→    57→
    58→    58→  /* ── Catalogue: online fetch with offline cache fallback ── */
    59→    59→  const loadCatalog = useCallback(async () => {
    60→    60→    try {
    61→    61→      const data = await getPosCatalog(businessId);
    62→    62→      const cache: CatalogCache = {
    63→    63→        businessId,
    64→    64→        fetchedAt: new Date().toISOString(),
    65→    65→        products: data.products.map((p) => ({
    66→    66→          id: p.id, name: p.name, sku: p.sku, barcodes: p.barcodes,
    67→    67→          department: p.department, type: p.type, trackBatches: p.trackBatches,
    68→    68→          sellingPrice: Number(p.sellingPrice), taxRate: Number(p.taxRate), baseUnit: p.baseUnit,
    69→    69→          conversions: p.conversions.map((c) => ({ fromUnit: c.fromUnit, toUnit: c.toUnit, factor: Number(c.factor) })),
    70→    70→          stock: Number(p.stock),
    71→    71→        })),
    72→    72→        business: data.business ? { ...data.business, defaultTaxRate: Number(data.business.defaultTaxRate) } : null,
    73→    73→        disclosedNote: "Offline snapshot — cloud stock may be out of date.",
    74→    74→      };
    75→    75→      setCatalog(cache);
    76→    76→      await saveCatalogCache(cache);
    77→    77→    } catch {
    78→    78→      const cached = await loadCatalogCache(businessId);
    79→    79→      if (cached) {
    80→    80→        setCatalog(cached);
    81→    81→        setOffline({ online: false });
    82→    82→        toast({ title: "Offline mode", description: "Using cached catalogue. Stock may be out of date." });
    83→    83→      } else {
    84→    84→        toast({ title: "Cannot load POS catalogue", description: "You are offline and no cached catalogue exists yet.", variant: "destructive" });
    85→    85→      }
    86→    86→    }
    87→    87→  }, [businessId, setOffline]);
    88→    88→
    89→    89→  useEffect(() => { void loadCatalog(); }, [loadCatalog]);
    90→    90→
    91→    91→  const loadShift = useCallback(async () => {
    92→    92→    try {
    93→    93→      const r = await getMyShift(businessId);
    94→    94→      setShift(r?.shift ? { id: r.shift.id, openingCash: 0 } : null);
    95→    95→      if (!r?.shift) setOpenShiftDialog(true);
    96→    96→    } catch {
    97→    97→      setShift(null);
    98→    98→    }
    99→    99→  }, [businessId]);
   100→   100→
   101→   101→  useEffect(() => { void loadShift(); }, [loadShift]);
   102→   102→
   103→   103→  /* ── Barcode / search ── */
   104→   104→  const filtered = useMemo(() => {
   105→   105→    if (!catalog) return [];
   106→   106→    const q = search.trim().toLowerCase();
   107→   107→    return catalog.products.filter((p) => {
   108→   108→      if (deptFilter !== "ALL" && p.department !== deptFilter && p.department !== "BOTH") return false;
   109→   109→      if (!q) return true;
   110→   110→      return p.name.toLowerCase().includes(q) || p.sku.toLowerCase().includes(q) || p.barcodes.some((b) => b.includes(q));
   111→   111→    });
   112→   112→  }, [catalog, search, deptFilter]);
   113→   113→
   114→   114→  function handleBarcodeScan(e: React.KeyboardEvent<HTMLInputElement>) {
   115→   115→    if (e.key !== "Enter") return;
   116→   116→    const code = search.trim();
   117→   117→    if (!code) return;
   118→   118→    const product = catalog?.products.find((p) => p.barcodes.includes(code)) ?? filtered[0];
   119→   119→    if (product) {
   120→   120→      handleAdd(product);
   121→   121→      setSearch("");
   122→   122→    } else {
   123→   123→      toast({ title: "Product not found", description: `No product matches "${code}"`, variant: "destructive" });
   124→   124→    }
   125→   125→  }
   126→   126→
   127→   127→  function handleAdd(p: CachedProduct) {
   128→   128→    if (online && p.stock <= 0 && !p.trackBatches) {
   129→   129→      toast({ title: "Out of stock", description: `${p.name} has no stock`, variant: "destructive" });
   130→   130→      return;
   131→   131→    }
   132→   132→    addToCart({ productId: p.id, name: p.name, department: p.department, soldUnit: p.baseUnit, unitPrice: p.sellingPrice, discountAmount: 0, stock: p.stock, trackBatches: p.trackBatches });
   133→   133→  }
   134→   134→
   135→   135→  /* ── Checkout ── */
   136→   136→  async function handleOpenShift() {
   137→   137→    try {
   138→   138→      await openShift(businessId, Number(openingFloat) || 0);
   139→   139→      setOpenShiftDialog(false);
   140→   140→      await loadShift();
   141→   141→      onShiftChanged?.();
   142→   142→      toast({ title: "Shift opened", description: `Opening float ₦${openingFloat}` });
   143→   143→    } catch (err) {
   144→   144→      toast({ title: "Could not open shift", description: err instanceof Error ? err.message : "Error", variant: "destructive" });
   145→   145→    }
   146→   146→  }
   147→   147→
   148→   148→  function resetPayment() {
   149→   149→    setPayDialog(false); setPaymentMethod("CASH"); setCashGiven(""); setTransferRef(""); setSplitCash(""); setSplitTransfer(""); setUseSplit(false);
   150→   150→  }
   151→   151→
   152→   152→  async function handleComplete() {
   153→   153→    if (!catalog || !cart.length) return;
   154→   154→    setCompleting(true);
   155→   155→    const idempotencyKey = newSaleKey();
   156→   156→
   157→   157→    const lines = cart.map((c) => ({ productId: c.productId, quantity: c.quantity, soldUnit: c.soldUnit, discountAmount: c.discountAmount }));
   158→   158→    const payments = useSplit
   159→   159→      ? [
   160→   160→          { method: "CASH" as const, amount: Number(splitCash) || 0 },
   161→   161→          { method: "TRANSFER" as const, amount: Number(splitTransfer) || 0, reference: transferRef || null },
   162→   162→        ].filter((p) => p.amount > 0)
   163→   163→      : paymentMethod === "CASH"
   164→   164→        ? [{ method: "CASH" as const, amount: Number(cashGiven) || total }]
   165→   165→        : [{ method: paymentMethod, amount: total, reference: transferRef || null }];
   166→   166→
   167→   167→    try {
   168→   168→      if (!online) {
   169→   169→        // Queue locally — do NOT pretend it's synced
   170→   170→        await queueSale({
   171→   171→          idempotencyKey, businessId, shiftId: shift?.id ?? null,
   172→   172→          lines, payments, transactionDiscount: 0,
   173→   173→          clientCreatedAt: new Date().toISOString(),
   174→   174→          status: "pending", attempts: 0,
   175→   175→        });
   176→   176→        const pending = await listQueuedSales(businessId);
   177→   177→        setOffline({ pendingCount: pending.filter((q) => q.status === "pending").length });
   178→   178→        clearCart();
   179→   179→        resetPayment();
   180→   180→        toast({ title: "Sale stored offline", description: "Will sync automatically when connection returns. Not yet in the cloud." });
   181→   181→        return;
   182→   182→      }
   183→   183→
   184→   184→      const result = await completeSale({
   185→   185→        businessId, shiftId: shift?.id ?? null, lines, payments,
   186→   186→        transactionDiscount: 0, idempotencyKey, clientCreatedAt: null, offline: false,
   187→   187→      });
   188→   188→      if (result?.ok) {
   189→   189→        if (result.shortfallWarnings?.length) setOfflineWarnings(result.shortfallWarnings);
   190→   190→        setReceipt(result.receipt as unknown as Receipt);
   191→   191→        clearCart();
   192→   192→        resetPayment();
   193→   193→        bumpRefresh();
   194→   194→        void loadCatalog(); // refresh stock
   195→   195→        toast({ title: `Sale ${result.number} completed`, description: `Total ${fmt(result.total)}${Number(result.changeGiven) > 0 ? ` • change ${fmt(result.changeGiven)}` : ""}` });
   196→   196→      }
   197→   197→    } catch (err) {
   198→   198→      const msg = err instanceof Error ? err.message : "Sale failed";
   199→   199→      if (msg.includes("INSUFFICIENT_STOCK")) {
   200→   200→        toast({ title: "Insufficient stock", description: msg.split(":")[1] ?? "Not enough stock", variant: "destructive" });
   201→   201→      } else if (msg.includes("PERMISSION_DENIED")) {
   202→   202→        toast({ title: "Not permitted", description: msg.replace("PERMISSION_DENIED: ", ""), variant: "destructive" });
   203→   203→      } else {
   204→   204→        toast({ title: "Sale failed", description: msg, variant: "destructive" });
   205→   205→      }
   206→   206→    } finally {
   207→   207→      setCompleting(false);
   208→   208→    }
   209→   209→  }
   210→   210→
   211→   211→  async function handleManualSync() {
   212→   212→    const result = await syncOfflineSales(businessId);
   213→   213→    const pending = await listQueuedSales(businessId);
   214→   214→    setOffline({ pendingCount: pending.filter((q) => q.status === "pending").length });
   215→   215→    toast({ title: "Sync complete", description: `${result.synced} synced, ${result.failed} failed, ${result.remaining} remaining` });
   216→   216→    void loadCatalog();
   217→   217→  }
   218→   218→
   219→   219→  const paymentTotal = useSplit ? (Number(splitCash) || 0) + (Number(splitTransfer) || 0) : Number(cashGiven) || total;
   220→   220→  const changeDue = Math.max(0, paymentTotal - total);
   221→   221→  const balanceDue = Math.max(0, total - paymentTotal);
   222→   222→
   223→   223→  return (
   224→   224→    <div className="space-y-4">
   225→   225→      {/* Header row */}
   226→   226→      <div className="flex flex-wrap items-center gap-2">
   227→   227→        <div>
   228→   228→          <h1 className="text-2xl font-bold tracking-tight">Point of Sale</h1>
   229→   229→          <p className="text-sm text-slate-500">{catalog?.business?.name ?? ""} • {shift ? "Shift open" : "No shift"}</p>
   230→   230→        </div>
   231→   231→        <div className="ml-auto flex items-center gap-2">
   232→   232→          {!online && <Badge variant="outline" className="border-amber-300 bg-amber-50 text-amber-700"><WifiOff className="h-3 w-3 mr-1" /> Offline — sales queue locally</Badge>}
   233→   233→          {online && offline.pendingCount > 0 && (
   234→   234→            <Button size="sm" variant="outline" onClick={() => void handleManualSync()}><CloudOff className="h-3.5 w-3.5 mr-1" /> Sync {offline.pendingCount} offline sale(s)</Button>
   235→   235→          )}
   236→   236→          {!shift && online && <Button size="sm" onClick={() => setOpenShiftDialog(true)}>Open shift</Button>}
   237→   237→        </div>
   238→   238→      </div>
   239→   239→
   240→   240→      {!online && (
   241→   241→        <Alert className="border-amber-300 bg-amber-50">
   242→   242→          <WifiOff className="h-4 w-4 text-amber-600" />
   243→   243→          <AlertDescription className="text-amber-800">
   244→   244→            Offline mode: catalogue snapshot from {catalog?.fetchedAt ? new Date(catalog.fetchedAt).toLocaleTimeString() : "—"}.
   245→   245→            Stock shown may be out of date; offline sales queue locally and sync when reconnected.
   246→   246→          </AlertDescription>
   247→   247→        </Alert>
   248→   248→      )}
   249→   249→      {offlineWarnings.length > 0 && (
   250→   250→        <Alert className="border-amber-300 bg-amber-50">
   251→   251→          <CircleAlert className="h-4 w-4 text-amber-600" />
   252→   252→          <AlertDescription className="text-amber-800">{offlineWarnings.join(" • ")}</AlertDescription>
   253→   253→        </Alert>
   254→   254→      )}
   255→   255→
   256→   256→      <div className="grid gap-4 lg:grid-cols-5">
   257→   257→        {/* Products */}
   258→   258→        <div className="lg:col-span-3 space-y-3">
   259→   259→          <div className="flex gap-2">
   260→   260→            <div className="relative flex-1">
   261→   261→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-slate-400" />
   262→   262→              <Input
   263→   263→                ref={barcodeRef}
   264→   264→                autoFocus
   265→   265→                placeholder="Scan barcode or search products… (Enter adds first match)"
   266→   266→                value={search}
   267→   267→                onChange={(e) => setSearch(e.target.value)}
   268→   268→                onKeyDown={handleBarcodeScan}
   269→   269→                className="pl-9"
   270→   270→              />
   271→   271→              {search && <button className="absolute right-2 top-2.5" onClick={() => setSearch("")}><X className="h-4 w-4 text-slate-400" /></button>}
   272→   272→            </div>
   273→   273→            <Select value={deptFilter} onValueChange={(v: "ALL" | "PHARMACY" | "SUPERMARKET") => setDeptFilter(v)}>
   274→   274→              <SelectTrigger className="w-40"><SelectValue /></SelectTrigger>
   275→   275→              <SelectContent>
   276→   276→                <SelectItem value="ALL">All departments</SelectItem>
   277→   277→                <SelectItem value="PHARMACY">Pharmacy</SelectItem>
   278→   278→                <SelectItem value="SUPERMARKET">Supermarket</SelectItem>
   279→   279→              </SelectContent>
   280→   280→            </Select>
   281→   281→          </div>
   282→   282→
   283→   283→          <ScrollArea className="h-[calc(100vh-320px)] min-h-72">
   284→   284→            <div className="grid grid-cols-2 sm:grid-cols-3 xl:grid-cols-4 gap-2 pr-2">
   285→   285→              {filtered.map((p) => (
   286→   286→                <button
   287→   287→                  key={p.id}
   288→   288→                  onClick={() => handleAdd(p)}
   289→   289→                  className={cn(
   290→   290→                    "text-left rounded-lg border p-2.5 bg-white hover:border-teal-500 hover:shadow-sm transition-all active:scale-[0.98]",
   291→   291→                    p.stock <= 0 && !p.trackBatches && "opacity-50"
   292→   292→                  )}
   293→   293→                >
   294→   294→                  <div className="text-xs font-medium text-slate-400 flex items-center justify-between">
   295→   295→                    <span>{p.department === "PHARMACY" ? "Pharmacy" : p.department === "BOTH" ? "Both" : "Market"}</span>
   296→   296→                    <span className={cn("font-semibold", p.stock <= 0 ? "text-red-500" : "text-slate-500")}>
   297→   297→                      {p.trackBatches ? "" : `${p.stock} ${p.baseUnit}`}
   298→   298→                    </span>
   299→   299→                  </div>
   300→   300→                  <div className="mt-1 text-sm font-semibold leading-snug line-clamp-2 min-h-9">{p.name}</div>
   301→   301→                  <div className="mt-1.5 flex items-center justify-between">
   302→   302→                    <span className="font-bold text-teal-700">{fmt(p.sellingPrice)}</span>
   303→   303→                    <Plus className="h-4 w-4 text-teal-600" />
   304→   304→                  </div>
   305→   305→                </button>
   306→   306→              ))}
   307→   307→              {!filtered.length && <p className="col-span-full text-center text-sm text-slate-400 py-10">No products match.</p>}
   308→   308→            </div>
   309→   309→          </ScrollArea>
   310→   310→        </div>
   311→   311→
   312→   312→        {/* Cart */}
   313→   313→        <Card className="lg:col-span-2 flex flex-col self-start sticky top-20">
   314→   314→          <CardContent className="p-4 flex flex-col gap-3">
   315→   315→            <div className="flex items-center justify-between">
   316→   316→              <h2 className="font-semibold">Basket ({cart.length})</h2>
   317→   317→              {cart.length > 0 && <Button size="sm" variant="ghost" className="text-red-500 h-7" onClick={clearCart}><Trash2 className="h-3.5 w-3.5" /> Clear</Button>}
   318→   318→            </div>
   319→   319→            <ScrollArea className="max-h-[42vh]">
   320→   320→              <div className="space-y-2">
   321→   321→                {cart.map((c) => (
   322→   322→                  <div key={c.productId} className="rounded-lg border p-2 space-y-1.5">
   323→   323→                    <div className="flex items-start justify-between gap-2">
   324→   324→                      <span className="text-sm font-medium leading-snug">{c.name}</span>
   325→   325→                      <button onClick={() => removeFromCart(c.productId)} aria-label={`Remove ${c.name}`}><X className="h-3.5 w-3.5 text-slate-400 hover:text-red-500" /></button>
   326→   326→                    </div>
   327→   327→                    <div className="flex items-center gap-2">
   328→   328→                      <div className="flex items-center border rounded-md">
   329→   329→                        <button className="px-2 py-1" onClick={() => updateCartQty(c.productId, c.quantity - 1)} aria-label="decrease"><Minus className="h-3 w-3" /></button>
   330→   330→                        <input
   331→   331→                          className="w-12 text-center text-sm outline-none"
   332→   332→                          value={c.quantity}
   333→   333→                          onChange={(e) => updateCartQty(c.productId, Number(e.target.value) || 0)}
   334→   334→                          inputMode="decimal"
   335→   335→                        />
   336→   336→                        <button className="px-2 py-1" onClick={() => updateCartQty(c.productId, c.quantity + 1)} aria-label="increase"><Plus className="h-3 w-3" /></button>
   337→   337→                      </div>
   338→   338→                      <span className="text-xs text-slate-500">{c.soldUnit} × {fmt(c.unitPrice)}</span>
   339→   339→                      <span className="ml-auto font-semibold text-sm">{fmt(c.quantity * c.unitPrice - c.discountAmount)}</span>
   340→   340→                    </div>
   341→   341→                    {canDiscountLine && (
   342→   342→                      <div className="flex items-center gap-1.5">
   343→   343→                        <Label className="text-[10px] text-slate-400">Discount ₦</Label>
   344→   344→                        <input
   345→   345→                          className="w-20 rounded border px-1.5 py-0.5 text-xs"
   346→   346→                          value={c.discountAmount || ""}
   347→   347→                          placeholder="0"
   348→   348→                          inputMode="decimal"
   349→   349→                          onChange={(e) => updateCartDiscount(c.productId, Number(e.target.value) || 0)}
   350→   350→                        />
   351→   351→                      </div>
   352→   352→                    )}
   353→   353→                  </div>
   354→   354→                ))}
   355→   355→                {!cart.length && <p className="text-sm text-slate-400 text-center py-10">Basket is empty — scan or tap a product.</p>}
   356→   356→              </div>
   357→   357→            </ScrollArea>
   358→   358→
   359→   359→            <div className="border-t pt-3 space-y-1.5">
   360→   360→              <div className="flex justify-between text-sm text-slate-500"><span>Items</span><span>{cart.reduce((a, c) => a + c.quantity, 0)}</span></div>
   361→   361→              <div className="flex justify-between text-lg font-bold"><span>Total</span><span className="text-teal-700">{fmt(total)}</span></div>
   362→   362→              <Button
   363→   363→                className="w-full bg-teal-600 hover:bg-teal-700 h-12 text-base"
   364→   364→                disabled={!cart.length || completing || (!online && !catalog)}
